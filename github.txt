	1. -------------------- Проверяем задан ли пользователь --------------------
		git config --get user.name
		git config --get user.email
		cat ~/.gitconfig (для того, чтобы посмотреть gitconfig фаил, где находятся глобальные переменные)
		ls -al ~/.ssh (Смотрим задан ли ssh ключ. Если задан, будет фаил с расширением pub)
		Если ssh ключ не задан, пишем команду: ssh-keygen -t ed25519 -C 'почта_регистрации_пользователя'
		После надо привязать ssh ключ к Github аккаунту. 
		git clone + сслыка на репозиторий (так проще всего подключиться к remote-репозиторию)

	2. -------------------- Если пользователь не задан --------------------
		git config --add --global user.name 'github_user_name'
		git config --add --global user.email 'github_user_email'

	-------------------- Создание репозтории и добавление --------------------

		git init
		git add README.md		
		git commit -m "first commit"
		git branch -M main
		git remote add origin https://github.com/VladimirKonstantinovich/-.git
		git push -u origin main


	-------------------- Добавление в существующую репозторию --------------------

		git remote add origin https://github.com/VladimirKonstantinovich/-.git
		git branch -M main
		git push -u origin main

	-------------------- Команды для работы с Git --------------------
		1. git add - добавляет фаил в промежуточную стадию (начинает смотреть изменения файла)
		2. git status - показывает статус файлов локальной репозитории git. 
		3. git reset - удаляет изменения в Staging area.
		4. git commit - добавление в коммита. (git commit -m 'Описание коммита' / эти коммиты формируются локально)
		5. git log - просмотр истории комитов.
		6. git push - отправка локальных коммитов в репозиторию
		7. git pull - получиние изменений из remote-репозитория
		8. git push origin название_ветки - внести изменения с ветки на которой находимся, в другую ветку.
		9. git stash -  перемещение правок в локальное хранилище 
		10. git stash pop - извлечение правок из локального хранилища
		11. git checkout -b branch_name - создание ветки с последующим переключением на неё.
		12. git merge - слияние веток с сохранением истории слияний
		13. git rebase - слияние веток с упрощенной линейной историей
		14. git revert - откат изменений к началу любого коммита
